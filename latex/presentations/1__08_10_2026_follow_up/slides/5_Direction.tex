\begin{frame}{Proposed research direction}


\begin{columns}[c]
    
    \begin{column}{0.48\textwidth}
    Current research:
        \begin{itemize}
            \item Multi-permeability magnetic cores have been demonstrated
            \item Different materials lead to different permeability regions
            \item Shown to influence flux distribution and DC-bias behaviour
        \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
    Proposed workflow:
        \begin{itemize}
        \item Simulate an inductor
        \item Vary $\mu$ in different regions
        \item Evaluate L, saturation, flux density, losses, DC-bias behaviour
        \item Optimize $\mu$ -distribution
        \item Can FDM reproduce it?
        \end{itemize}
    \end{column}

\end{columns}

\end{frame}
